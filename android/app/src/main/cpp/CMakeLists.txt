cmake_minimum_required(VERSION 3.22)
project(image_border_codecs LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(FetchContent)
include(ExternalProject)
set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(ENABLE_STATIC ON CACHE BOOL "" FORCE)
set(WITH_TURBOJPEG OFF CACHE BOOL "" FORCE)
set(WITH_TOOLS OFF CACHE BOOL "" FORCE)
set(WITH_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(jpegturbo URL https://github.com/libjpeg-turbo/libjpeg-turbo/archive/refs/tags/3.1.2.tar.gz URL_HASH SHA256=560f6338b547544c4f9721b18d8b87685d433ec78b3c644c70d77adad22c55e6)
FetchContent_GetProperties(jpegturbo)
if(NOT jpegturbo_POPULATED)
  FetchContent_Populate(jpegturbo)
endif()
set(jpegturbo_BINARY_DIR ${CMAKE_CURRENT_BINARY_DIR}/jpeg-build)
file(MAKE_DIRECTORY ${jpegturbo_BINARY_DIR})
ExternalProject_Add(jpeg_external SOURCE_DIR ${jpegturbo_SOURCE_DIR} BINARY_DIR ${jpegturbo_BINARY_DIR}
  CMAKE_ARGS -DCMAKE_TOOLCHAIN_FILE=${CMAKE_TOOLCHAIN_FILE} -DCMAKE_MAKE_PROGRAM=${CMAKE_MAKE_PROGRAM} -DANDROID_ABI=${ANDROID_ABI} -DANDROID_PLATFORM=${ANDROID_PLATFORM}
    -DCMAKE_BUILD_TYPE=Release -DCMAKE_POSITION_INDEPENDENT_CODE=ON -DENABLE_SHARED=OFF -DENABLE_STATIC=ON -DWITH_TURBOJPEG=OFF -DWITH_TOOLS=OFF -DWITH_TESTS=OFF
  BUILD_COMMAND ${CMAKE_COMMAND} --build . --target jpeg-static --parallel 2
  INSTALL_COMMAND "" BUILD_BYPRODUCTS ${jpegturbo_BINARY_DIR}/libjpeg.a)
add_library(jpeg-static STATIC IMPORTED GLOBAL)
set_target_properties(jpeg-static PROPERTIES IMPORTED_LOCATION ${jpegturbo_BINARY_DIR}/libjpeg.a)
add_dependencies(jpeg-static jpeg_external)
set(PNG_SHARED OFF CACHE BOOL "" FORCE)
set(PNG_STATIC ON CACHE BOOL "" FORCE)
set(PNG_TESTS OFF CACHE BOOL "" FORCE)
set(PNG_TOOLS OFF CACHE BOOL "" FORCE)
find_library(ANDROID_ZLIB z REQUIRED)
set(ZLIB_LIBRARY ${ANDROID_ZLIB} CACHE FILEPATH "" FORCE)
set(ZLIB_INCLUDE_DIR ${CMAKE_SYSROOT}/usr/include CACHE PATH "" FORCE)
FetchContent_Declare(png URL https://github.com/pnggroup/libpng/archive/refs/tags/v1.6.50.tar.gz URL_HASH SHA256=71158e53cfdf2877bc99bcab33641d78df3f48e6e0daad030afe9cb8c031aa46)
FetchContent_MakeAvailable(png)
FetchContent_Declare(uhdr URL https://api.github.com/repos/google/libultrahdr/tarball/v1.4.0 URL_HASH SHA256=bdd3f7f43a94c355c734959deef0d3d0eb396deed0abb342fba9514e5a944d5d)
FetchContent_GetProperties(uhdr)
if(NOT uhdr_POPULATED)
  FetchContent_Populate(uhdr)
endif()
file(GLOB UHDR_SOURCES ${uhdr_SOURCE_DIR}/lib/src/*.cpp ${uhdr_SOURCE_DIR}/third_party/image_io/src/**/*.cc)
add_library(uhdr_core STATIC ${UHDR_SOURCES})
target_compile_definitions(uhdr_core PRIVATE UHDR_WRITE_XMP UHDR_WRITE_ISO)
target_include_directories(uhdr_core PRIVATE ${uhdr_SOURCE_DIR}/third_party/image_io/src/modp_b64/modp_b64)
add_dependencies(uhdr_core jpeg_external)
target_include_directories(uhdr_core PUBLIC ${uhdr_SOURCE_DIR} PRIVATE ${uhdr_SOURCE_DIR}/lib/include ${uhdr_SOURCE_DIR}/third_party/image_io/includes ${uhdr_SOURCE_DIR}/third_party/image_io/includes/modp_b64 ${uhdr_SOURCE_DIR}/third_party/image_io/src/modp_b64 ${jpegturbo_SOURCE_DIR}/src ${jpegturbo_SOURCE_DIR} ${jpegturbo_BINARY_DIR})
target_link_libraries(uhdr_core PRIVATE jpeg-static log)
add_library(imagecodecs SHARED codecs.cpp)
target_compile_options(imagecodecs PRIVATE -O2)
add_dependencies(imagecodecs jpeg_external)
target_include_directories(imagecodecs PRIVATE ${jpegturbo_SOURCE_DIR}/src ${jpegturbo_SOURCE_DIR} ${jpegturbo_BINARY_DIR} ${png_SOURCE_DIR} ${png_BINARY_DIR} ${uhdr_SOURCE_DIR})
target_link_libraries(imagecodecs PRIVATE jpeg-static png_static uhdr_core jnigraphics android log z)
target_link_options(imagecodecs PRIVATE -Wl,-z,max-page-size=16384)
